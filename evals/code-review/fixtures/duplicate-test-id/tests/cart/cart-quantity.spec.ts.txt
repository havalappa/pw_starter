import { test, expect } from '../../fixtures';
import { PRODUCTS } from '../../data/products';

test.describe('Cart quantity', () => {
  test(
    'C01 – should keep a single cart line when the same product is added twice',
    { tag: '@regression' },
    async ({ homePage, cartPage }) => {
      await homePage.goto();
      await homePage.searchFor(PRODUCTS.pliers.keyword);
      await homePage.addProductToCart(PRODUCTS.pliers.name);
      await homePage.goto();
      await homePage.searchFor(PRODUCTS.pliers.keyword);
      await homePage.addProductToCart(PRODUCTS.pliers.name);
      await cartPage.goto();
      await expect(cartPage.cartItems).toHaveCount(1);
    },
  );
});
